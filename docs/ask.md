<script>
document.getElementById('ai-ask-btn').addEventListener('click', async () => {
    const questionInput = document.getElementById('ai-question');
    const loadingDiv = document.getElementById('ai-loading');
    const responseContainer = document.getElementById('ai-response-container');
    const answerDiv = document.getElementById('ai-answer');
    const sourcesDiv = document.getElementById('ai-sources');
    const errorDiv = document.getElementById('ai-error');
    
    const question = questionInput.value.trim();
    if (!question) return;

    // Reset UI
    loadingDiv.style.display = 'block';
    responseContainer.style.display = 'none';
    errorDiv.style.display = 'none';
    answerDiv.textContent = '';
    sourcesDiv.innerHTML = '';
    
    try {
        const response = await fetch('https://ai-backend-cms-phy-det-analysis-ask.app.cern.ch/chat', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({ question: question })
        });
        
        if (!response.ok) {
            throw new Error(`Server returned status ${response.status}: ${response.statusText}`);
        }
        
        const data = await response.json();
        
        // Show response
        loadingDiv.style.display = 'none';
        responseContainer.style.display = 'block';
        answerDiv.textContent = data.answer || "No answer returned.";
        
        if (data.sources && data.sources.length > 0) {
            sourcesDiv.innerHTML = '<strong>Sources:</strong><ul>' + 
                data.sources.map(s => `<li>${s}</li>`).join('') + '</ul>';
        } else {
            sourcesDiv.innerHTML = '<em>No specific sources used yet (RAG pending).</em>';
        }
        
    } catch (error) {
        loadingDiv.style.display = 'none';
        errorDiv.style.display = 'block';
        errorDiv.textContent = 'Failed to connect to AI backend: ' + error.message;
    }
});
</script>
